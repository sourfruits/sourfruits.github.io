# data/precursors.txt — the hand-edited source for the Precursors graph.

# Running  node scripts/build-precursors.mjs  compiles this into
# data/precursors.json, which the site actually reads. Edit THIS file, not the
# JSON (the JSON is generated and overwritten on every build). Same relationship
# as data/posts/*.md → data/posts.json.
#
# ── Format cheat-sheet — one node using every possible field ──────────────────
#
# ### template-node-id            # "### id" starts a node; it runs until the next "###"
# label: Display Name             # required
# kind: film                      # required — free string (film, book, person, platform, class, album, …)
# creator: Some Person            # optional — the Author/Director byline
# connections:                    # optional — each item on its own indented "- " line
#   - to: other-node-id, relationship: influence, note: how the two connect
#   - to: another-node-id         # relationship + note optional; a bare "to" is a plain line
#                                 # relationship, if given, must be: adaptation, influence, authorship, thematic
# discovered:                     # optional — each item on its own indented "- " line
#   - source: some-node-id, aware: 2023-04, engaged: 2026-05, thread: 70s paranoia, note: the story, mechanism: platform-letterboxd
#   - aware: 2024                 # every field optional, but an entry can't be empty
#                                 # dates are YYYY, YYYY-MM, or YYYY-MM-DD
#
# Notes:
#   • Blank lines and lines starting with "#" are ignored.
#   • A value containing a comma must be quoted:  note: "Girl, so confusing".
#   • label + kind are required; a discovery entry needs at least one field.
# ─────────────────────────────────────────────────────────────────────────────
